\section*{a) Definición de los términos}
\begin{itemize}

    \item \textbf{Buffer Manager}
    \begin{itemize}
        \item Componente del DBMS encargado de administrar la memoria principal utilizada para almacenar páginas de datos.
        \item Reduce accesos al disco mediante el uso de memoria RAM.
        \item Gestiona la carga, reemplazo y escritura de páginas.
    \end{itemize}

    \item \textbf{Buffer Pool}
    \begin{itemize}
        \item Área de memoria RAM administrada por el Buffer Manager.
        \item Contiene páginas copiadas temporalmente desde el disco.
        \item Está dividido en frames.
    \end{itemize}

    \item \textbf{Frames}
    \begin{itemize}
        \item Espacios individuales dentro del Buffer Pool.
        \item Cada frame puede almacenar una sola página.
        \item Funcionan como contenedores de páginas en memoria.
    \end{itemize}

    \item \textbf{Pages}
    \begin{itemize}
        \item Unidad básica de almacenamiento en un DBMS.
        \item Contienen registros, índices y metadatos.
        \item Tienen tamaño fijo (por ejemplo 4KB u 8KB).
    \end{itemize}

    \item \textbf{Dirty Pages}
    \begin{itemize}
        \item Páginas modificadas en memoria pero no escritas aún al disco.
        \item Se marcan como ``dirty'' después de una actualización.
        \item Permiten mejorar rendimiento reduciendo escrituras inmediatas.
    \end{itemize}

    \item \textbf{Page Replacement}
    \begin{itemize}
        \item Proceso de seleccionar qué página eliminar del buffer cuando está lleno.
        \item Busca optimizar el uso de memoria y minimizar accesos a disco.
    \end{itemize}

    \item \textbf{Cache Hit}
    \begin{itemize}
        \item Ocurre cuando la página solicitada ya se encuentra en el Buffer Pool.
        \item Permite acceso rápido sin leer desde disco.
    \end{itemize}

    \item \textbf{Cache Miss}
    \begin{itemize}
        \item Ocurre cuando la página solicitada no está en memoria.
        \item Requiere cargar la página desde disco.
    \end{itemize}

    \item \textbf{Algoritmos del Buffer Manager}
    \begin{itemize}
        \item \textbf{FIFO (First In First Out):}
        elimina la primera página que ingresó al buffer.

        \item \textbf{LRU (Least Recently Used):}
        elimina la página menos utilizada recientemente.

        \item \textbf{MRU (Most Recently Used):}
        elimina la página usada más recientemente.

        \item \textbf{LFU (Least Frequently Used):}
        elimina la página menos utilizada históricamente.

        \item \textbf{Clock Algorithm:}
        aproximación eficiente de LRU utilizando bits de referencia.
    \end{itemize}

\end{itemize}

\section*{c) Desarrollar un programa en C++ que simule paso a paso el algoritmo LRU.}

\begin{lstlisting}[caption=Implementación del Storage Engine]
#include <vector>
#include <algorithm>
#include <iostream>

using namespace std;

int main() {
    cout << "num access, num frames:\n";
    int n, f; cin >> n >> f;
    
    cout << n << " frames input:\n";
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<int> mem; // acts as a queue of frames, front is LRU, back is MRU
    int faults = 0;

    for (int i = 0; i < n; i++) {
        cout << "Access " << i + 1 << ": Page " << a[i] << " -> ";
        auto it = find(mem.begin(), mem.end(), a[i]);
        
        if (it != mem.end()) {
            cout << "HIT.          ";
            // move the accessed page to the end
            int page = *it;
            mem.erase(it);
            mem.push_back(page);
        } else {
            cout << "PAGE FAULT.   ";
            // page fault
            faults++;
            if (mem.size() < (size_t)f) {
                mem.push_back(a[i]);
            } else {
                // remove the least recently used page (the one at the front)
                mem.erase(mem.begin());
                mem.push_back(a[i]);
            }
        }

        cout << "Frames: [";
        for (size_t j = 0; j < mem.size(); j++) {
            cout << mem[j] << (j == mem.size() - 1 ? "" : ", ");
        }
        cout << "]" << endl;
    }

    cout << "pages fails: " << faults << endl;

    return 0;
}
\end{lstlisting}
